import numpy as np
board = [
[7,8,0,4,0,0,1,2,0],
[6,0,0,0,7,5,0,0,9],
[0,0,0,6,0,1,0,7,8],
[0,0,7,0,4,0,2,6,0],
[0,0,1,0,5,0,9,3,0],
[9,0,4,0,6,0,0,0,5],
[0,7,0,3,0,0,0,1,2],
[1,2,0,0,0,7,4,0,0],
[0,4,9,2,0,6,0,0,7]
]

#demande une librairie mais permet de trouver d'autres éventuelles solutions

def possible(row, col, number):
  global board
  for i in range(0,9):
    if board[row][i] == number:
      return False
  for i in range(0,9):
    if board[i][col] == number:
      return False

  x0 = col //3 *3
  y0 = row //3 *3
  for i in range (0,3):
    for j in range (0,3):
      if board[y0+i][x0+j] ==number:
        return False
  return True

def solve():
  global board
  for row in range(0,9):
    for col in range (0,9):
      if board[row][col]==0:
        for number in range (1,10):
          if possible(row,col,number):
            board[row][col] = number
            solve()
            board[row][col]= 0
        return

  print(np.matrix(board))
  input("more possible solutions")
solve()
