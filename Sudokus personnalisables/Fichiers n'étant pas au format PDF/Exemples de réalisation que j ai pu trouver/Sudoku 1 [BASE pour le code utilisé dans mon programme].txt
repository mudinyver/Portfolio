#avantage visuelle

def solve (bo):
  find = find_empty(bo) #voir fonction find_empty(bo) trouvable plus bas.
  if not find :
    return True #si le programme ne trouve plus de 0 alors cela veut dire que le sudoku est terminé.
  else :
    row, col = find #le rpogramme va récupérer les coordonnées du 0.

  for i in range(1,10):
    if valid(bo, i, (row, col)): 
      bo[row][col] = i #si un nombre est valide il est enregistré

      if solve(bo):
       return True #solution trouvée
      bo[row][col] =0
  return False  #valeur annulée

def valid(bo, num, pos):
  #check ligne
  for i in range (len(bo[0])):
    if bo[pos[0]][i] == num and pos [1] != i:
      return False
  #check colonne
  for i in range (len(bo[0])):
    if bo[i][pos[1]] == num and pos [0] != i:
      return False
  #check régions
  region_x= pos[1] // 3
  region_y= pos[0] // 3

  for i in range(region_y*3, region_y*3 + 3): #étendue de la région en lignes
    for j in range(region_x * 3, region_x*3 + 3): #étendue de la région en colonnes
      if bo[i][j] ==num and (i,j) != pos:
        return False
  return True

def print_grille(bo):
  for i in range (len(bo)):
    if i % 3 ==0 and i !=0:
      print("- - - - - - - - - - - -") #affiche des lignes horizontales si i est égal à 3 ou 6

    for j in range(len(bo[0])):
      if j % 3 ==0 and j!=0:
        print(" | ", end="") #affiche des lignes verticales si j est égal à 3 ou 6
      if j==8: #si arrivé à fin de ligne
        print (bo[i][j])
      else:
        print(str(bo[i][j]) + " ", end="")

def find_empty(bo): 
  for i in range (len(bo)):
    for j in range (len(bo[0])):
      if bo[i][j] == 0:
        return (i, j)
  return None #si la fonction trouve un 0 elle renvoie son emplacement
