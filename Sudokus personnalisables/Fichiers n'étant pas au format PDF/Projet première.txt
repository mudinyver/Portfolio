def SupprNonAlphabet(Suppr): #fonction permettant de supprimer des lettres qui ne sont pas présentes dans l'alphabet

      del Suppr[s]
      s=s-1
    s=s+1
  Suppr = "".join(Suppr)  #Transforme la liste en chaine de caractères
  return(Suppr)
