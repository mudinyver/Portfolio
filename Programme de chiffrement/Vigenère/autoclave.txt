choix_alphabet_autoclave=-1
while choix_alphabet_autoclave<1 or choix_alphabet_autoclave>3:
  choix_alphabet_autoclave=int(input("Quel version de l'halphabet voulez-vous utiliser ?\n""1  Version classique\n""2  Variante clavier français (AZERTY...)\n""3  Variante clavier anglais (QWERTY...)\n"))
def creer_dico():
    dico={}
    if choix_alphabet_autoclave ==1:
      majuscules="ABCDEFGHIJKLMNOPQRSTUVWXYZ"
    elif choix_alphabet_autoclave ==2:
      majuscules="AZERTYUIOPQSDFGHJKLMWXCVBN"
    else:
      majuscules="QWERTYUIOPASDFGHJKLZXCVBNM"
    majuscules=list(majuscules)
    for i in range(26):
        dico[i]=majuscules[i]
    return(dico)

def creer_dico_inv():
    dico={}
    if choix_alphabet_autoclave ==1:
      majuscules="ABCDEFGHIJKLMNOPQRSTUVWXYZ"
    elif choix_alphabet_autoclave ==2:
      majuscules="AZERTYUIOPQSDFGHJKLMWXCVBN"
    else:
      majuscules="QWERTYUIOPASDFGHJKLZXCVBNM"
    majuscules=list(majuscules)
    for i in range(26):
        dico[majuscules[i]]=i
    return(dico)

def autoclave(texte,cle):
  texte=texte.upper()   #Transforme les minuscules en majuscules
  texte=list(texte)
  cle=list(cle)
  print(list(cle))
  cle_texte=list(cle)+list(texte)
  decrypt=[]
  dic=creer_dico()
  dic_inv=creer_dico_inv()
  nbcar=len(texte)
  nbcle=len(cle_texte)
  j=0
  for i in range(0,nbcar):
      k=dic_inv[texte[i]]+dic_inv[cle_texte[j]]
      if k>25:
          k=k-26
      if j==nbcle-1:
          j=0
      else:
          j=j+1
      decrypt.append(dic[k])
  return(decrypt)

print(autoclave("BONJOUR","CLE"))
