choix_alphabet_Beaufort=-1
while choix_alphabet_Beaufort<1 or choix_alphabet_Beaufort>3:
  choix_alphabet_Beaufort=int(input("Quel version de l'halphabet voulez-vous utiliser ?\n""1  Version classique\n""2  Variante clavier français (AZERTY...)\n""3  Variante clavier anglais (QWERTY...)\n"))
def creer_dico():
    dico={}
    if choix_alphabet_Beaufort ==1:
      majuscules="ABCDEFGHIJKLMNOPQRSTUVWXYZ"
    elif choix_alphabet_Beaufort ==2:
      majuscules="AZERTYUIOPQSDFGHJKLMWXCVBN"
    else:
      majuscules="QWERTYUIOPASDFGHJKLZXCVBNM"
    majuscules=list(majuscules)
    for i in range(26):
        dico[i]=majuscules[i]
    return(dico)

def creer_dico_inv():
    dico={}
    if choix_alphabet_Beaufort ==1:
      majuscules="ABCDEFGHIJKLMNOPQRSTUVWXYZ"
    elif choix_alphabet_Beaufort ==2:
      majuscules="AZERTYUIOPQSDFGHJKLMWXCVBN"
    else:
      majuscules="QWERTYUIOPASDFGHJKLZXCVBNM"
    majuscules=list(majuscules)
    for i in range(26):
        dico[majuscules[i]]=i
    return(dico)

def Beaufort(texte,cle):
    texte=texte.upper()   #Transforme les minuscules en majuscules
    texte=list(texte)
    cle=list(cle)
    decrypt=[]
    dic=creer_dico()
    dic_inv=creer_dico_inv()
    nbcar=len(texte)
    nbcle=len(cle)
    j=0
    for i in range(0,nbcar):
        k=dic_inv[cle[j]]-dic_inv[texte[i]]
        if k<1:
            k=k+26
        if j==nbcle-1:
            j=0
        else:
            j=j+1
        decrypt.append(dic[k])
    return(decrypt)

print(Beaufort("BONJOUR","CLE"))
