minuscules=list(minuscules)

def creat_dico1(minuscules,decal_lettres):
  dico1={}
  for i in range(lenm1):
    decal_lettres=decal_lettres+1  #Pour que le décalage augmente à chaque caractère
    if decal_lettres>lenm1:
      decal_lettres=decal_lettres-26
      if i+decal_lettres>=lenm1:
       dico1[minuscules[i]]=minuscules[i+decal_lettres-26]
      else:
        pass
    elif i+decal_lettres>=lenm1:
      dico1[minuscules[i]]=minuscules[i+decal_lettres-26]
    else:
      dico1[minuscules[i]]=minuscules[i+decal_lettres]
  return(dico1)

def codage(texte, dico_min, dico_maj, dico_chiffres):
  texte_c=[]
  for lettre in texte:
    if (lettre in dico_min)==True:
      texte_c.append(dico_min[lettre]) #Problème de ce code le dictionnaire qui rend une incrémentation différente de deux mêmes lettre difficile
    elif (lettre in dico_maj)==True:
      texte_c.append(dico_maj[lettre])
    elif (lettre in dico_chiffres)==True:
      texte_c.append(dico_chiffres[lettre])
    else:
      texte_c.append(lettre)
  return (texte_c)
