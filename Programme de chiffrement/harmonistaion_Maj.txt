txt=input("Veuillez entrez le texte que vous voulez (majuscules et minuscules uniquement).")
majuscules="ABCDEFGHIJKLMNOPQRSTUVWXYZ"

def harmonistaion_Maj(Harmoni_Maj):
  Harmoni_Maj=Harmoni_Maj.upper()   #Transforme les majuscules en minuscule
  Harmoni_Maj=list(Harmoni_Maj)
  i=0
  while i<len(Harmoni_Maj):
    if (Harmoni_Maj[i]=='Î' or Harmoni_Maj[i]=='Ï' or Harmoni_Maj[i]=='Í' or Harmoni_Maj[i]=='Ì'):
      Harmoni_Maj[i]='I'
    elif (Harmoni_Maj[i]=='Û' or Harmoni_Maj[i]=='Ü' or Harmoni_Maj[i]=='Ù' or Harmoni_Maj[i]=='Ú' ):
      Harmoni_Maj[i]='U'
    elif (Harmoni_Maj[i]=='Ö' or Harmoni_Maj[i]=='Ô' or Harmoni_Maj[i]=='Ó' or Harmoni_Maj[i]=='Ò' or Harmoni_Maj[i]=='Õ'):
      Harmoni_Maj[i]='O'
    elif (Harmoni_Maj[i]=='Ñ'):
      Harmoni_Maj[i]='N'
    elif (Harmoni_Maj[i]=='Ê' or Harmoni_Maj[i]=='Ë' or Harmoni_Maj[i]=='É' or Harmoni_Maj[i]=='È'):
      Harmoni_Maj[i]='E'
    elif (Harmoni_Maj[i]=='Ç'):
      Harmoni_Maj[i]='C'
    elif (Harmoni_Maj[i]=='À'or Harmoni_Maj[i]=='Á' or Harmoni_Maj[i]=='Ä'or Harmoni_Maj[i]=='Â' or Harmoni_Maj[i]=='Ã' or Harmoni_Maj[i]=='Å'):
      Harmoni_Maj[i]='A'
    elif (Harmoni_Maj[i] in majuscules)==False:
      del Harmoni_Maj[i]
      i=i-1
    else:
      pass
    i=i+1
  Harmoni_Maj = "".join(Harmoni_Maj)  #Transforme la liste en chaine de caractères
  return(Harmoni_Maj)
mot_code=harmonistaion_Maj(txt)
print(mot_code)
