lenmROT13=len(majuscules)
decalROT13=13
def creat_dicoROT13(majuscules,decalROT13):
  dicoROT13={}
  for j in range(lenmROT13):
    if j+decalROT13>=lenmROT13:
      dicoROT13[majuscules[j]]=majuscules[j+decalROT13-26]
    else:
      dicoROT13[majuscules[j]]=majuscules[j+decalROT13]
  return(dicoROT13)

def codage(texte, dico_ROT13):
  texte_c=[]
  texte=texte.upper()   #Transforme les minuscules en majuscules
  for lettre in texte:
    if (lettre in dico_ROT13)==False:
      texte_c.append(lettre)
    else:
      texte_c.append(dico_ROT13[lettre])
  return (texte_c)

mot_code=codage("ABCd", creat_dicoROT13(majuscules,decalROT13))
