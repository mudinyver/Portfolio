minuscules="abcdefghijklmnopqrstuvwxyz"
minuscules=list(minuscules)
lenm=len(minuscules)
decal=3
def creat_dico(minuscules,decal):
  dico={}
  for i in range(lenm):
    if i+decal>=lenm:
      dico[minuscules[i]]=minuscules[i+decal-26]
    else:
      dico[minuscules[i]]=minuscules[i+decal]
  return(dico)

def codage(texte,dico):
  texte_c=[]
  for lettre in texte:
    if (lettre in dico)==False:
      texte_c.append(lettre)
    else:
      texte_c.append(dico[lettre])
  return (texte_c)

mot_code=codage("bonjour à vous",creat_dico(minuscules,decal))
mot_code="".join(mot_code)
print(mot_code)
